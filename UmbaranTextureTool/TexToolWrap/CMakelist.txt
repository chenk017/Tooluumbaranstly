cmake_minimum_required(VERSION 3.10)

project(textoolwrap)

set(CMAKE_CXX_STANDARD 17)

add_library(
    textoolwrap
    SHARED
    textoolwrap.cpp
)


target_include_directories(
    textoolwrap
    PRIVATE
    .
    ./PVRTexLib/Include
    ./ispc/include
    ./crunch/inc
)

    textoolwrap
    PRIVATE
    TEXTTOOLWRAP_EXPORTS
)


set_target_properties(
    textoolwrap
    PROPERTIES
    OUTPUT_NAME "textoolwrap"
)
